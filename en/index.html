def pedir(id_serie, desde, hasta, token):
    url = f"{API}/{id_serie}/datos/{desde}/{hasta}"
    req = urllib.request.Request(
        url, headers={"Bmx-Token": token}
    )
    with urllib.request.urlopen(req, timeout=60) as resp:
        cuerpo = json.load(resp)
    datos = cuerpo["bmx"]["series"][0].get("datos") or []
    filas = []
    for d in datos:
        valor = d["dato"].replace(",", "")
        if valor in ("N/E", ""):
            continue
        filas.append((d["fecha"], float(valor)))
    return filas
